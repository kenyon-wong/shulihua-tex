\newcommand*\fitfig[1]{%
  \sbox\shulihua@box{#1}%
  \dimen@=\dimexpr\textheight-4\baselineskip\relax
  \Gscale@div\@tempa{\dimen@}{\dimexpr\ht\shulihua@box+\dp\shulihua@box\relax}%
  \Gscale@div\@tempb{\linewidth}{\wd\shulihua@box}%
  \ifdim\@tempb\p@<\@tempa\p@\let\@tempa\@tempb\fi
  \ifdim\@tempa\p@<\p@\scalebox{\@tempa}{\usebox\shulihua@box}%
  \else\usebox{\shulihua@box}%
  \fi
}

\long\def\@makecaption#1#2{%
  \vskip\abovecaptionskip
  \sbox\@tempboxa{{\small #2}}%
  \ifdim \wd\@tempboxa >\hsize
    {\small #2\par}%
  \else
    \global \@minipagefalse
    \hb@xt@\hsize{\hfil\small #2\hfil}%
  \fi
  \vskip\belowcaptionskip
}
